-- 1. [JOINS]
SELECT e.name, j.job_title, d.dept_name 
FROM Employees e 
JOIN Jobs j ON e.job_id = j.job_id 
JOIN Departments d ON e.dept_id = d.dept_id;

-- 2. [SUBQUERY]
SELECT name 
FROM Employees 
WHERE salary > (SELECT AVG(salary) FROM Employees e JOIN Departments d ON e.dept_id = d.dept_id WHERE d.dept_name = 'IT');

-- 3. [GROUP BY]
SELECT d.dept_name, SUM(e.salary) as total_salary 
FROM Employees e 
JOIN Departments d ON e.dept_id = d.dept_id 
GROUP BY d.dept_name;

-- 4. [HAVING]
SELECT d.dept_name 
FROM Departments d 
JOIN Employees e ON d.dept_id = e.dept_id 
GROUP BY d.dept_name 
HAVING COUNT(e.emp_id) > 2 AND AVG(e.salary) > 70000;

-- 5. [SELF JOIN]
SELECT e1.name AS employee_name, e2.name AS manager_name 
FROM Employees e1 
LEFT JOIN Employees e2 ON e1.manager_id = e2.emp_id;

-- 6. [ORDER BY]
SELECT name, salary 
FROM Employees 
ORDER BY salary DESC 
LIMIT 3;

-- 7. [JOINS + WHERE]
SELECT e.name 
FROM Employees e 
JOIN Jobs j ON e.job_id = j.job_id 
JOIN Departments d ON e.dept_id = d.dept_id 
WHERE j.job_title = 'Developer' AND d.location = 'San Francisco';

-- 8. [GROUP BY]
SELECT EXTRACT(YEAR FROM hire_date) AS hire_year, COUNT(*) AS num_employees 
FROM Employees 
GROUP BY EXTRACT(YEAR FROM hire_date);

-- 9. [HAVING]
SELECT j.job_title 
FROM Jobs j 
JOIN Employees e ON j.job_id = e.job_id 
GROUP BY j.job_title 
HAVING MAX(e.salary) < 100000;

-- 10. [SUBQUERY]
SELECT name 
FROM Employees 
WHERE dept_id = (SELECT dept_id FROM Employees WHERE name = 'David') AND name != 'David';

-- 11. [MULTI-TABLE JOIN]
SELECT d.dept_name, e.name
FROM Departments d 
LEFT JOIN Employees e ON d.dept_id = e.dept_id;

-- 12. [COMPLEX FILTER]
SELECT name 
FROM Employees 
WHERE EXTRACT(YEAR FROM hire_date) BETWEEN 2018 AND 2020 
AND salary > 60000;

-- 13. [CORRELATED SUBQUERY]
SELECT e.name 
FROM Employees e 
WHERE e.salary > (SELECT AVG(salary) FROM Employees e2 WHERE e2.dept_id = e.dept_id);

-- 14. [JOINS]
SELECT j.job_title, e.salary 
FROM Employees e 
JOIN Jobs j ON e.job_id = j.job_id 
WHERE e.emp_id = 504;

-- 15. [GROUP BY]
SELECT manager_id, MIN(salary) AS min_salary 
FROM Employees 
WHERE manager_id IS NOT NULL 
GROUP BY manager_id;

-- 16. [SUBQUERY]
SELECT dept_name 
FROM Departments 
WHERE dept_id NOT IN (SELECT e.dept_id FROM Employees e JOIN Jobs j ON e.job_id = j.job_id WHERE j.job_title = 'Analyst');

-- 17. [AGGREGATION]
SELECT MAX(salary) - MIN(salary) AS salary_difference 
FROM Employees;

-- 18. [JOIN + ORDER BY]
SELECT e.name, e.hire_date 
FROM Employees e 
JOIN Departments d ON e.dept_id = d.dept_id 
ORDER BY d.dept_name ASC, e.salary DESC;

-- 19. [HAVING]
SELECT manager_id 
FROM Employees 
WHERE manager_id IS NOT NULL 
GROUP BY manager_id 
HAVING COUNT(emp_id) > 2;

-- 20. [SUBQUERY]
SELECT name, salary 
FROM Employees 
ORDER BY salary DESC 
LIMIT 1 OFFSET 2;

-- 21. [LEFT JOIN]
SELECT j.job_title, COUNT(e.emp_id) AS num_employees 
FROM Jobs j 
LEFT JOIN Employees e ON j.job_id = e.job_id 
GROUP BY j.job_title;

-- 22. [JOIN + LIKE]
SELECT e.name 
FROM Employees e 
JOIN Departments d ON e.dept_id = d.dept_id 
WHERE d.dept_name = 'Finance' AND e.name LIKE '%i%';
